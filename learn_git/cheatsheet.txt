1. git add <filename> (або "." для всіх файлів) - добавити файли у staging area (готуємо до коміту)
2. git commit -m "message" - створити коміт (точку в історії)
3. git push - Відправити зміни з локального у віддалений репозиторій
4. git pull - Витягнути зміни з віддаленого репозиторію у локальний (відразу з merge)
5. git fetch - Витягнути зміни з віддаленого репозиторію у локальний (без merge)
6. git merge - Злиття комітів/гілок між собою
7. git branch - глянути всі гілки
8. git branch <назва> - створити нову гілку
9. git checkout <назва> - перейти на нову гілку (або git switch)
10. git checkout -B <назва> - створити і перейти на нову гілку
